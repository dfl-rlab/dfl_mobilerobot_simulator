cmake_minimum_required(VERSION 3.16)
project(robot_sensors_description)

find_package(ament_cmake REQUIRED)
find_package(gz_sim_vendor REQUIRED)
find_package(gz-sim8 REQUIRED)
find_package(gz_rendering_vendor REQUIRED)
find_package(gz-rendering8 REQUIRED)
find_package(gz_transport_vendor REQUIRED)
find_package(gz-transport13 REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(yaml-cpp REQUIRED)

add_library(dome_lidar_gpu SHARED src/dome_lidar_gpu.cpp)
target_compile_features(dome_lidar_gpu PRIVATE cxx_std_17)
target_compile_options(dome_lidar_gpu PRIVATE -Wall -Wextra -Wpedantic)
target_include_directories(dome_lidar_gpu PRIVATE include)
target_link_libraries(dome_lidar_gpu gz-sim8::gz-sim8 gz-rendering8::gz-rendering8 yaml-cpp)
ament_target_dependencies(dome_lidar_gpu rclcpp sensor_msgs)
add_library(robot_reset SHARED src/robot_reset.cpp)
target_compile_features(robot_reset PRIVATE cxx_std_17)
target_link_libraries(robot_reset gz-sim8::gz-sim8 gz-transport13::gz-transport13)
install(TARGETS robot_reset dome_lidar_gpu LIBRARY DESTINATION lib)
ament_environment_hooks(
  env-hooks/gazebo_plugin_path.dsv.in
  env-hooks/gazebo_model_path.dsv.in)
install(DIRECTORY xacro rviz config meshes DESTINATION share/${PROJECT_NAME})
ament_package()
